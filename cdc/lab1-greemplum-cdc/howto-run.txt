CREATE DATABASE my_database;

\c my_database;


CREATE TABLE t1 (
    id int PRIMARY KEY, 
    yourmessage text
) DISTRIBUTED BY (id);

CREATE TABLE t2 (
    id int PRIMARY KEY, 
    yourmessage text
) DISTRIBUTED BY (id);



CREATE PUBLICATION mypub FOR TABLE t1, t2;


CREATE FEED myfeed
  CONNECTION 'host=localhost port=5672 user=admin password=password vhost=/ exchange=amq.topic'
  PUBLICATION mypub
  WITH (
    target = rabbitmq,
    plugin = wal2json,
    target_options='{"ssl","false"}'
  );



SELECT oid, subname FROM pg_subscription WHERE subname = 'myfeed';





SELECT 
    s.oid, 
    s.subname, 
    d.datname AS target_database, 
    current_database() AS current_database
FROM pg_subscription s
JOIN pg_database d ON s.subdbid = d.oid
WHERE s.subname = 'myfeed';
